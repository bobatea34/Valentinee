<script>
const noBtn = document.getElementById("noBtn");
const message = document.getElementById("message");
const yesBtn = document.getElementById("yesBtn");
const popup = document.getElementById("popup");

function escapeNo() {
  const x = Math.random() * (window.innerWidth - 100);
  const y = Math.random() * (window.innerHeight - 60);
  noBtn.style.left = x + "px";
  noBtn.style.top = y + "px";
  message.style.display = "block";
}

noBtn.addEventListener("mouseover", escapeNo);
noBtn.addEventListener("click", escapeNo);

// YES opens fake popup
yesBtn.addEventListener("click", () => {
  popup.style.display = "flex";
});

function closePopup() {
  popup.style.display = "none";
}

function confirmYes() {
  popup.style.display = "none";
  document.getElementById("mainPage").style.display = "none";
  document.getElementById("lovePage").style.display = "block";
  launchConfetti();
}

// Hearts
setInterval(() => {
  const heart = document.createElement("div");
  heart.className = "heart";
  heart.innerText = "❤️";
  heart.style.left = Math.random() * 100 + "vw";
  document.body.appendChild(heart);
  setTimeout(() => heart.remove(), 5000);
}, 250);

// Confetti
function launchConfetti() {
  for (let i = 0; i < 120; i++) {
    const confetti = document.createElement("div");
    confetti.className = "confetti";
    confetti.style.left = Math.random() * 100 + "vw";
    confetti.style.backgroundColor =
      ["#ff1744","#ff4081","#f50057","#ff80ab"][Math.floor(Math.random()*4)];
    confetti.style.animationDuration = (Math.random()*2 + 2) + "s";
    document.body.appendChild(confetti);
    setTimeout(() => confetti.remove(), 3000);
  }
}
</script>
